let
    var calcular = fn a b . a + b 
in
    let
        var processarDados = fn x y . calcular(x, y) - 5
    in
        describe("Suite Avancada: Propagacao de Erros na AST",
            
            test("1. Falha de Logica (Controle)", 
                 expect(processarDados(10, 20), 99)
            ),
            
            test("2. Exception Bubbling (Erro de Cast Profundo)", 
                 toThrow(processarDados(10, "texto invalido"))
            ),
            
            test("3. Sobrevivencia do Interpretador", 
                 expect(calcular(5, 5), 10)
            )
        )
